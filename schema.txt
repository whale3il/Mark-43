Enum CurrencyCode {
  USD   // US Dollar (triggers sovereign settlement wallet auto-provisioning)
  EUR   // Euro
  GBP   // British Pound Sterling
  CHF   // Swiss Franc
  NGN   // Nigerian Naira
}

Enum AccountType {
  checking        // High-liquidity operating account
  savings         // High-yield reserve vault (APY accrual)
  investment      // Capital syndicate / private equity allocation
  multicurrency   // Multi-currency cross-border vault
}

Enum TransactionStatus {
  settled         // Finalized & cleared funds
  pending         // Under clearing / AML verification
  processing      // Dispatched across correspondent network
}

Enum CardTier {
  Obsidian_Black_Metal         // Physical obsidian metal card ($250k limit)
  Titanium_Precision_Virtual   // Ephemeral / virtual tokenized card ($50k limit)
  Corporate_Platinum          // Multi-signatory corporate debit
}

Enum KycTier {
  Tier_1          // Basic verification
  Tier_2          // Identity & address verified
  Tier_3          // Sovereign Private Client (BVN, passport, source of wealth verified)
}

# Table: accounts
# Maps to: Account model
# CRITICAL BUSINESS RULE:
# Any account where currency = 'USD' MUST automatically provision and link 
# a unique sovereign cryptographic settlement wallet address (EIP-55 format, 
# e.g., '0x8b32A46F0C9d71A12B8025e16738Eb2463FbcE39') with is_wallet_provisioned = true.
# Non-USD accounts (EUR, GBP, CHF, NGN) have wallet_address = null and is_wallet_provisioned = false.
Table accounts {
  id                     String        [PK, default: cuid/uuid]
  user_id                String        [FK -> users.id, Cascade Delete]
  name                   String        [Not Null, e.g., "Premier Liquidity Checking"]
  type                   AccountType   [Default: checking]
  account_number         String        [Unique, Not Null, e.g., "8940 3120 4821"]
  routing_number         String        [Default: "021000089"]
  iban                   String?       [e.g., "GB29 NWBK 6016 1331 9268 19"]
  swift_bic              String?       [e.g., "UBSWCHZH"]
  balance                Decimal(18,2) [Default: 0.00]
  available_balance      Decimal(18,2) [Default: 0.00]
  currency               CurrencyCode  [Default: USD]
  interest_rate          Decimal(5,2)? [Optional, e.g., 4.85 for Treasury Vault]
  color_theme            String        [Default: "zinc"]
  
  # Sovereign Wallet Provisioning Fields:
  wallet_address         String?       [Unique, EIP-55 42-char hex string, e.g. 0x8b32...]
  is_wallet_provisioned  Boolean       [Default: false]

  created_at             DateTime      [Default: now()]
  updated_at             DateTime      [Auto-update: now()]

  # Relationships:
  transactions           transactions[]
}

# Table: transactions

  id                     String            [PK, default: cuid/uuid]
  account_id             String            [FK -> accounts.id, Cascade Delete]
  date                   String            [ISO Date: YYYY-MM-DD]

  status                 TransactionStatus [Default: settled]

  payment_method         String            [Default: "Fedwire Realtime"]
  reference_number       String            [Unique, e.g., "WRE-948201"]
  notes                  String?           [Memo / Transfer purpose]
  recipient_wallet       String?           [Optional settlement wallet address]
  created_at             DateTime          [Default: now()]
}

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Custom Types
DO $$ BEGIN
  CREATE TYPE currency_code AS ENUM ('USD', 'EUR', 'GBP', 'CHF', 'NGN');
EXCEPTION
  WHEN duplicate_object THEN null;
END $$;

DO $$ BEGIN
  CREATE TYPE account_type AS ENUM ('checking', 'savings', 'investment', 'multicurrency');
EXCEPTION
  WHEN duplicate_object THEN null;
END $$;

DO $$ BEGIN
  CREATE TYPE transaction_status AS ENUM ('settled', 'pending', 'processing');
EXCEPTION
  WHEN duplicate_object THEN null;
END $$;

-- 1. Users Table

# 5. RECOMMENDED API ENDPOINTS & PAYLOAD SHAPES
# ------------------------------------------------------------------------------
# The frontend connects to the following endpoints via `src/services/bankingService.ts`
# and `src/services/authService.ts`:
#
# Authentication:
#   POST /api/auth/register       -> { email, username, firstName, lastName, password }
#   POST /api/auth/verify-email   -> { email, code: "123456" }
#   POST /api/auth/resend-code    -> { email }
#   POST /api/auth/login          -> { email, password }
#   GET  /api/auth/me             -> Returns current User & Profile
#   POST /api/auth/logout         -> Terminates session
#
# Banking & Ledger:
#   GET  /api/banking/dashboard   -> Consolidated { accounts, transactions, cards, bills, goals, beneficiaries }
#   GET  /api/banking/accounts    -> Returns all user accounts
#   POST /api/banking/accounts    -> { name, type, currency, initialDeposit, colorTheme }
#                                    (Auto-provisions wallet_address if currency == 'USD')
#   GET  /api/banking/transactions-> Returns transaction history
#   POST /api/banking/transfers   -> { sourceAccountId, beneficiaryId, amount, speed, memo }
#                                    (Debits account balance and inserts transaction)
#   GET  /api/banking/cards       -> Returns Obsidian & Titanium cards
#   PATCH/api/banking/cards/:id   -> { isFrozen, contactlessEnabled, onlinePaymentsEnabled, monthlyLimit }
#   POST /api/banking/cards       -> Issues new virtual / physical card
#   GET  /api/banking/beneficiaries -> Counterparty address book
#   POST /api/banking/beneficiaries -> Adds verified counterparty
#   DELETE /api/banking/beneficiaries/:id -> Removes counterparty
#   GET  /api/banking/bills       -> Scheduled bills & executive invoices
#   POST /api/banking/bills/:id/pay -> Pays bill and settles transaction
#   PATCH/api/banking/bills/:id/autopay -> Toggles autopay status
#   GET  /api/banking/goals       -> Capital goals & private wealth milestones
#   POST /api/banking/goals/:id/deposit -> Allocates funds into goal
#   GET  /api/banking/sessions    -> Active security devices
#   DELETE /api/banking/sessions/:id -> Revokes remote session
#
# Real-Time Event Stream (SSE or Supabase Realtime):
#   GET  /api/realtime
#   Events emitted: 'account:created', 'account:updated', 'transaction:created', 
#                   'transfer:executed', 'card:updated', 'bill:paid', 'goal:updated'
